## 差分转储源码（probe.py；只调用既有模块，不新造判定）

"""T207-FIX-SCORE：默认参数（P1 口径）差分转储 —— 同一批输入喂给 old/new 两套源码。

用法：PYTHONPATH=<包根> python -X utf8 -B <本文件> --out <json>
判定：两次转储逐条相等 = 默认口径 0 分歧（本次修改只在显式传 allowed_number_keys 时生效）。
"""

from __future__ import annotations

import argparse
import json
from pathlib import Path

import mktkb
from mktkb.evaluation.dataset import CaseSpec
from mktkb.evaluation.gold import ResolvedGold
from mktkb.evaluation.p2_scoring import _resolved_without_index, normalize_product_outcome
from mktkb.evaluation.scorers import EvalCitation, EvalOutcome, score_case

ROOT = Path(".")
FIXTURES = (
    ROOT / "fixtures" / "evaluation" / "b0.jsonl",
    ROOT / "fixtures" / "evaluation" / "badcase.jsonl",
    ROOT / "fixtures" / "evaluation" / "regression.jsonl",
)
RUNS = (
    ROOT / "docs/reports/OBS-EVAL/P2/T207/p2-t207-r1-20261111-01",
    ROOT / "docs/reports/OBS-EVAL/P2/T207/p2-t207-r2-20261111-01",
    ROOT / "docs/reports/OBS-EVAL/P2/T207/p2-t207-r3-20261111-01",
)
DOC_CODES = ROOT / "docs/reports/OBS-EVAL/P2/T207/cases/document-codes.json"

#: 引用 locator 自带数字（Q09 式整行记录）：默认口径必须无视它，opt-in 才用它。
CITATION = EvalCitation(
    doc_code="D01",
    unit_key=None,
    unit_ids=("unit-1",),
    locator={
        "sheet": "店家信息",
        "row": 5,
        "cells": [{"display": "010-00000001"}, {"display": "示例市示例路18号（模拟）"}],
    },
    text="",
    kind="active",
)


def gold(case: CaseSpec) -> ResolvedGold:
    return ResolvedGold(
        case_id=case.case_id,
        requires_ocr=bool(case.requires_ocr),
        skipped=False,
        sources=(),
        unit_ids=(),
        relevant_unit_ids=(),
    )


def variants(case: CaseSpec) -> dict[str, str]:
    echo = case.reference_answer or "；".join(case.expected_values) or case.question
    return {
        "echo_gold": echo,
        "echo_plus_number": echo + "。另有 99 件在库。",
        "refusal": "没有找到相关资料，无法回答。",
        "echo_plus_citation_numbers": echo + "。电话 010-00000001，地址示例市示例路18号。",
    }


def fixture_records() -> list[dict]:
    records: list[dict] = []
    for path in FIXTURES:
        if not path.is_file():
            continue
        for line in path.read_text(encoding="utf-8").splitlines():
            if not line.strip():
                continue
            case = CaseSpec.from_dict(json.loads(line))
            for name, text in variants(case).items():
                outcome = EvalOutcome(
                    kind=str(case.expected_kind or "answer"),
                    text=text,
                    citations=(CITATION,),
                    path="rag",
                )
                result = score_case(case, gold(case), outcome)
                records.append(
                    {
                        "id": f"{path.stem}:{case.case_id}:{name}",
                        "result": result.to_dict(),
                    }
                )
    return records


def run_records() -> list[dict]:
    codes = json.loads(DOC_CODES.read_text(encoding="utf-8"))
    records: list[dict] = []
    for run in RUNS:
        if not run.is_dir():
            continue
        cases = {
            case.case_id: case
            for case in (
                CaseSpec.from_dict(json.loads(line))
                for line in (run / "dataset.jsonl").read_text(encoding="utf-8").splitlines()
                if line.strip()
            )
        }
        for line in (run / "cases.jsonl").read_text(encoding="utf-8").splitlines():
            if not line.strip():
                continue
            row = json.loads(line)
            case = cases.get(row.get("case_id"))
            turns = row.get("turns") or ()
            if case is None or not turns or not isinstance(turns[-1].get("product"), dict):
                continue
            outcome = normalize_product_outcome(turns[-1], document_codes=codes)
            result = score_case(
                case, _resolved_without_index(case), outcome, cross_explicit_space=True
            )
            records.append(
                {"id": f"{run.name}:{case.case_id}", "result": result.to_dict()}
            )
    return records


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", required=True)
    args = parser.parse_args()
    payload = {
        "module": mktkb.__file__,
        "fixture_records": fixture_records(),
        "run_records": run_records(),
    }
    Path(args.out).write_text(json.dumps(payload, ensure_ascii=False, indent=1), encoding="utf-8")
    print(
        json.dumps(
            {
                "module": payload["module"],
                "fixture_records": len(payload["fixture_records"]),
                "run_records": len(payload["run_records"]),
            },
            ensure_ascii=False,
        )
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
