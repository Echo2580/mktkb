import importlib.util, json, sys
from pathlib import Path

ROOT = Path(r"D:\GPT项目\市场部AI知识库")
sys.path.insert(0, str(ROOT / "backend" / "src"))
sys.path.insert(0, str(ROOT / "backend" / "tests"))

spec = importlib.util.spec_from_file_location("t_h2diag", ROOT / "backend/tests/evaluation/test_p2_holdout_scorecard.py")
mod = importlib.util.module_from_spec(spec)
sys.modules["t_h2diag"] = mod
spec.loader.exec_module(mod)

card, specs = mod._diag_card()
print("=== SYNTHETIC DIAGNOSTICS ===")
print(json.dumps(card["diagnostics"], ensure_ascii=False, indent=2, sort_keys=True))
print("CARD", json.dumps({k: card[k] for k in ("total","executed","passed","failed","errors")}, ensure_ascii=False))
print("BLOCKS", [b["code"] for b in card["blocks"]])
mod.runner._assert_holdout_scorecard_payload(card, mod._synthetic_dataset(specs))
print("AUDIT ok")
