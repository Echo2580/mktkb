```python
import json
from mktkb.evaluation.dataset import CaseSpec, load_cases
from mktkb.evaluation.gold import ResolvedGold
from mktkb.evaluation.scorers import EvalCitation, EvalOutcome, score_case

KINDS = ("answer", "no_answer", "not_found", "refusal", "clarification", "conflict", "ambiguous", "error")
TEXTS = ("", "B区1号货架第二层 2026-10-30 16:00 共12件", "该事项资料未收录，无法回答")
CITATION = EvalCitation(doc_code="D01", unit_key="sheet:物品位置:row:8", unit_ids=("u1",))
PAGINATION = {"total": 1, "returned": 1, "has_more": False, "next_page": None}
ROOT = "D:/GPT项目/市场部AI知识库/fixtures/evaluation/"

cases = [case for name in ("b0.jsonl", "badcase.jsonl", "regression.jsonl") for case in load_cases(ROOT + name)]
with open(ROOT + "holdout-questions.jsonl", encoding="utf-8") as handle:
    for line in handle:
        if not line.strip():
            continue
        row = json.loads(line)
        cases.append(
            CaseSpec(
                case_id=str(row["case_id"]),
                suite="holdout",
                split="holdout",
                stratum=str(row["stratum"]),
                business_date=str(row["business_date"]),
                question=str(row["question"]),
                expected_kind=str(row["expected_kind"]),
                labeler=str(row["labeler"]),
                label_version=str(row["label_version"]),
                should_refuse=bool(row["should_refuse"]),
                requires_ocr=bool(row["requires_ocr"]),
            )
        )

rows = []
for case in cases:
    gold = ResolvedGold(case_id=case.case_id, requires_ocr=bool(case.requires_ocr), skipped=False,
                        sources=(), unit_ids=(), relevant_unit_ids=())
    for kind in KINDS:
        for text in TEXTS:
            result = score_case(case, gold, EvalOutcome(kind=kind, text=text, citations=(CITATION,), pagination=PAGINATION))
            rows.append({"case_id": case.case_id, "kind": kind, "text_index": TEXTS.index(text), **result.to_dict()})
print(json.dumps({"cases": len(cases), "comparisons": len(rows), "rows": rows}, ensure_ascii=False, sort_keys=True))
```
