```python
"""u-controls 的 Windows 解码缺陷：同一 pytest 调用在旧/新两种 subprocess 参数下的差异。

旧形态：subprocess.run(text=True) —— Windows 用 locale(cp936) 解码 pytest 的 UTF-8 输出，
       读取线程抛 UnicodeDecodeError（父进程 stderr 出现线程异常；exit 仍为 0）。
新形态：encoding="utf-8", errors="replace" + 子进程 PYTHONIOENCODING=utf-8（T207-FIX-A-R1 修法）。

用法：python u-controls-decode-probe.py {old|new}   （stderr 用 shell 重定向单独留证）
"""
import os
import subprocess
import sys
from pathlib import Path

REPO = Path(__file__).resolve().parents[6]
target = REPO / "backend" / "tests" / "acceptance" / "test_t09_mutation.py"
common = dict(cwd=str(REPO), capture_output=True, text=True,
              timeout=1200,
              args=[sys.executable, "-m", "pytest", str(target), "-q", "-p", "no:cacheprovider"])

mode = sys.argv[1] if len(sys.argv) > 1 else "old"
args = common.pop("args")
if mode == "old":
    done = subprocess.run(args, **common)
else:
    done = subprocess.run(args, encoding="utf-8", errors="replace",
                          env={**os.environ, "PYTHONIOENCODING": "utf-8"}, **common)
stdout = done.stdout or ""
print(f"mode={mode} exit={done.returncode} stdout_chars={len(stdout)} "
      f"has_passed={'passed' in stdout}")
print("stdout_tail=" + stdout.strip().splitlines()[-1][:120] if stdout.strip() else "stdout_tail=<empty>")
```
