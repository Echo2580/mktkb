import json
import sys
from pathlib import Path

from mktkb.evaluation.scorers import check_value, normalize_text, _value_pattern

ROOT = Path(sys.argv[1])
OUT = Path(sys.argv[2])
FIX = ROOT / "fixtures"

rows = []
def add(case_id, stratum, suite, expected, forbidden):
    rows.append({"case_id": case_id, "stratum": stratum, "suite": suite,
                 "expected": list(expected or ()), "forbidden": list(forbidden or ())})

for name in ["b0.jsonl", "badcase.jsonl", "regression.jsonl", "refusal-controls-2026-11-10.jsonl"]:
    p = FIX / "evaluation" / name
    if not p.exists():
        continue
    for line in p.read_text(encoding="utf-8").splitlines():
        if not line.strip():
            continue
        r = json.loads(line)
        add(r.get("case_id"), r.get("stratum"), r.get("suite"),
            r.get("expected_values"), r.get("forbidden_values"))

accepted = json.loads((FIX / "acceptance" / "questions.json").read_text(encoding="utf-8"))
items = accepted.get("questions") if isinstance(accepted, dict) else accepted
for q in items or ():
    sc = q.get("scenario") or {}
    add(q.get("id") or q.get("question_id"), sc.get("category") or q.get("category"),
        q.get("group"), sc.get("expected_values"), sc.get("forbidden_values"))

# 1) 清单：期望/禁止值里含内部空白的
whitespace = []
for r in rows:
    for kind in ("expected", "forbidden"):
        for v in r[kind]:
            if any(c.isspace() for c in v.strip()):
                whitespace.append((r["case_id"], r["stratum"], kind, v))

# 2) 差分：默认口径（P1 路径用的就是默认）对"值 x 文本变体"逐个比对
def variants(value):
    v = value
    return {
        "exact": v,
        "in_sentence": f"结果：{v}。",
        "in_sentence_ws": f"结果：{v.replace(' ', '  ')}。",   # 多余空白
        "fullwidth": "结果：" + "。" if False else None,
        "prefix0": f"0{v}",
        "suffix0": f"{v}0",
        "drop_tail": v[:-1] if len(v) > 1 else v,
        "drop_head": v[1:] if len(v) > 1 else v,
        "double": f"{v}{v}",
        "unrelated": "资料中没有相关记录。",
        "empty": "",
    }

results = {}
pairs = 0
for r in rows:
    for kind in ("expected", "forbidden"):
        for v in r[kind]:
            for label, text in variants(v).items():
                if text is None:
                    continue
                key = f'{r["case_id"]}|{kind}|{label}'
                results[key] = check_value(text, v)
                pairs += 1

patterns = {v: _value_pattern(normalize_text(v).replace(" ", "")).pattern
            for r in rows for kind in ("expected", "forbidden") for v in r[kind]}

payload = {
    "case_rows": len(rows),
    "whitespace_values": whitespace,
    "pairs_checked": pairs,
    "results": results,
    "patterns": patterns,
}
OUT.write_text(json.dumps(payload, ensure_ascii=False, indent=1, sort_keys=True), encoding="utf-8")
print(f"cases={len(rows)} whitespace_values={len(whitespace)} pairs={pairs} patterns={len(patterns)}")
for w in whitespace:
    print("  WS", w)
